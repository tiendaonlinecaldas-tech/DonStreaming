async function obtenerTextoLista(url) { // Array de proxies para saltarse el bloqueo de HTTP y CORS const proxies = [ u => `https://corsproxy.io/?${encodeURIComponent(u)}`, u => `https://api.allorigins.win/raw?url=${encodeURIComponent(u)}` ]; // 1. Intentar acceso directo primero try { const res = await fetch(url); if (res.ok) return await res.text(); } catch (e) {} // 2. Intentar a través de proxies si falla por HTTP/CORS for (let proxyFn of proxies) { try { const res = await fetch(proxyFn(url)); if (res.ok) return await res.text(); } catch (e) {} } throw new Error('No se pudo conectar al servidor IPTV. Verifica que la lista esté activa.'); }
